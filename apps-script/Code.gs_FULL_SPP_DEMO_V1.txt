/**
 * Jalankan SEKALI setelah Code.gs dipasang.
 *
 * Function ini akan:
 * 1. Membuat spreadsheet database demo otomatis.
 * 2. Membuat sheet STUDENTS, BILLS, ACTIVITY_LOG.
 * 3. Mengisi 1 santri demo + tagihan contoh.
 *
 * Setelah selesai, lihat Execution Log untuk URL spreadsheet.

function setupSppPaymentDemoV1() {

  const props =
    PropertiesService
      .getScriptProperties();


  let spreadsheetId =
    String(
      props.getProperty(
        SPP_DEMO_CONFIG.spreadsheetProperty
      ) || ''
    ).trim();


  let spreadsheet;


  if (spreadsheetId) {

    spreadsheet =
      SpreadsheetApp.openById(
        spreadsheetId
      );

  } else {

    spreadsheet =
      SpreadsheetApp.create(
        'SPP Payment Demo Database'
      );


    spreadsheet.setSpreadsheetTimeZone(
      SPP_DEMO_CONFIG.timezone
    );


    spreadsheetId =
      spreadsheet.getId();


    props.setProperty(
      SPP_DEMO_CONFIG.spreadsheetProperty,
      spreadsheetId
    );
  }


  createOrResetDemoSheets_(
    spreadsheet
  );


  seedDemoData_(
    spreadsheet
  );


  console.log(
    '=============================='
  );

  console.log(
    'SPP PAYMENT DEMO V1 READY'
  );

  console.log(
    'Spreadsheet: ' +
    spreadsheet.getUrl()
  );

  console.log(
    'Spreadsheet ID: ' +
    spreadsheetId
  );

  console.log(
    'NEXT: pastikan Script Property ' +
    SPP_DEMO_CONFIG.paymentSecretProperty +
    ' sudah diisi.'
  );
}

/**
 * Public GET API untuk frontend GitHub Pages.
 *
 * Endpoint:
 * ?action=ping
 * ?action=getDashboard
 * ?action=createPayment&bill_id=...
 * ?action=refreshStatus&bill_id=...
 *
 * Browser TIDAK mengirim nominal.
 */
function doGet(
  e
) {

  try {

    const params =
      e && e.parameter
        ? e.parameter
        : {};


    const action =
      String(
        params.action || ''
      ).trim();


    let result;


    switch (action) {

      case 'ping':

        result = {
          success: true,
          message: 'SPP Payment Demo backend aktif.',
          data: {
            app_name:
              SPP_DEMO_CONFIG.appName,
            environment:
              'SANDBOX'
          }
        };

        break;


      case 'getDashboard':

        result =
          getSppDemoDashboard_();

        break;


      case 'createPayment':

        result =
          createSppDemoPayment_(
            params.bill_id
          );

        break;


      case 'refreshStatus':

        result =
          refreshSppDemoPaymentStatus_(
            params.bill_id
          );

        break;


      default:

        result = {
          success: false,
          code: 'UNKNOWN_ACTION',
          message: 'Action tidak dikenali.'
        };
    }


    return outputSppDemoResponse_(
      result,
      params.callback
    );

  } catch (error) {

    return outputSppDemoResponse_(
      {
        success: false,
        code: 'SPP_DEMO_ERROR',
        message:
          error &&
          error.message
            ? error.message
            : 'Terjadi kesalahan pada backend demo.'
      },
      e &&
      e.parameter
        ? e.parameter.callback
        : ''
    );
  }
}

/* ============================================================
 * SECTION 5 — CREATE PAYMENT
 * ============================================================
 */

/**
 * Browser hanya mengirim bill_id.
 *
 * Amount, customer dan description dibaca dari Spreadsheet.
 */
function createSppDemoPayment_(
  billId
) {

  const cleanBillId =
    String(
      billId || ''
    ).trim();


  if (!cleanBillId) {

    throw new Error(
      'bill_id wajib diisi.'
    );
  }


  const lock =
    LockService.getScriptLock();


  lock.waitLock(
    20000
  );


  try {

    const spreadsheet =
      getSppDemoSpreadsheet_();


    const billResult =
      findBillRow_(
        spreadsheet,
        cleanBillId
      );


    if (!billResult) {

      throw new Error(
        'Tagihan tidak ditemukan.'
      );
    }


    const bill =
      billResult.data;


    if (
      bill.local_status === 'PAID'
    ) {

      return {
        success: false,
        code: 'BILL_ALREADY_PAID',
        message: 'Tagihan ini sudah lunas.'
      };
    }


    /*
     * Kalau link sebelumnya masih aktif, cukup gunakan ulang.
     * Tidak membuat PAYMENT_REQUEST baru.
     */
    if (
      bill.payment_link &&
      (
        bill.local_status ===
          'PAYMENT_CREATED' ||
        bill.local_status ===
          'OPEN'
      )
    ) {

      return {
        success: true,
        message: 'Payment link aktif digunakan kembali.',
        data: {
          bill_id:
            bill.bill_id,
          payment_link:
            bill.payment_link,
          local_status:
            bill.local_status,
          reused:
            true
        }
      };
    }


    const student =
      getStudentById_(
        spreadsheet,
        bill.student_id
      );


    if (!student) {

      throw new Error(
        'Data santri untuk tagihan tidak ditemukan.'
      );
    }


    const currentAttempt =
      Number(
        bill.attempt_count || 0
      ) + 1;


    const runId =
      getOrCreateDemoRunId_();


    const referenceNumber =
      [
        'SPPDEMO',
        runId,
        bill.bill_id,
        'A' + currentAttempt
      ].join(
        '-'
      );


    const paymentResult =
      createPaymentRequestToEngine_(
        {
          referenceNumber:
            referenceNumber,

          customerId:
            student.student_id,

          customerName:
            student.name,

          description:
            bill.description +
            ' - ' +
            bill.period,

          amount:
            Number(
              bill.amount
            ),

          notes:
            'SPP Payment Demo Sandbox'
        }
      );


    const paymentToken =
      extractTokenFromPaymentLink_(
        paymentResult.payment_link
      );


    const nowIso =
      new Date()
        .toISOString();


    updateBillRow_(
      billResult,
      {
        local_status:
          'PAYMENT_CREATED',

        payment_request_id:
          paymentResult.payment_request_id || '',

        payment_reference:
          referenceNumber,

        payment_link:
          paymentResult.payment_link || '',

        payment_token:
          paymentToken,

        token_expires_at:
          paymentResult.token_expires_at || '',

        attempt_count:
          currentAttempt,

        last_checked_at:
          '',

        updated_at:
          nowIso
      }
    );


    appendActivityLog_(
      spreadsheet,
      {
        action:
          'CREATE_PAYMENT',
        bill_id:
          bill.bill_id,
        detail:
          referenceNumber,
        status:
          'PAYMENT_CREATED'
      }
    );


    return {
      success: true,
      message: 'Link pembayaran berhasil dibuat.',
      data: {
        bill_id:
          bill.bill_id,
        payment_request_id:
          paymentResult.payment_request_id || '',
        payment_link:
          paymentResult.payment_link,
        local_status:
          'PAYMENT_CREATED',
        token_expires_at:
          paymentResult.token_expires_at || '',
        reused:
          false
      }
    };

  } finally {

    lock.releaseLock();
  }
}


/* ============================================================
 * SECTION 6 — REFRESH PAYMENT STATUS
 * ============================================================
 */

function refreshSppDemoPaymentStatus_(
  billId
) {

  const cleanBillId =
    String(
      billId || ''
    ).trim();


  if (!cleanBillId) {

    throw new Error(
      'bill_id wajib diisi.'
    );
  }


  const spreadsheet =
    getSppDemoSpreadsheet_();


  const billResult =
    findBillRow_(
      spreadsheet,
      cleanBillId
    );


  if (!billResult) {

    throw new Error(
      'Tagihan tidak ditemukan.'
    );
  }


  const bill =
    billResult.data;


  if (
    bill.local_status === 'PAID'
  ) {

    return {
      success: true,
      message: 'Tagihan sudah lunas.',
      data: {
        bill_id:
          bill.bill_id,
        local_status:
          'PAID',
        paid_at:
          bill.paid_at || ''
      }
    };
  }


  if (!bill.payment_token) {

    return {
      success: true,
      message: 'Belum ada payment link untuk tagihan ini.',
      data: {
        bill_id:
          bill.bill_id,
        local_status:
          bill.local_status || 'UNPAID'
      }
    };
  }


  const secureStatus =
    getSecurePaymentStatusFromEngine_(
      bill.payment_token
    );


  const engineStatus =
    String(
      secureStatus.status || ''
    ).toUpperCase();


  let localStatus =
    'PAYMENT_CREATED';


  if (
    engineStatus === 'PAID' ||
    engineStatus === 'SUCCESS'
  ) {

    localStatus =
      'PAID';

  } else if (
    engineStatus === 'EXPIRED' ||
    engineStatus === 'CANCELLED'
  ) {

    localStatus =
      'EXPIRED';
  }


  const nowIso =
    new Date()
      .toISOString();


  const patch = {

    local_status:
      localStatus,

    last_checked_at:
      nowIso,

    updated_at:
      nowIso
  };


  if (
    localStatus === 'PAID'
  ) {

    patch.paid_at =
      secureStatus.paid_at ||
      nowIso;
  }


  updateBillRow_(
    billResult,
    patch
  );


  appendActivityLog_(
    spreadsheet,
    {
      action:
        'REFRESH_STATUS',
      bill_id:
        bill.bill_id,
      detail:
        engineStatus || 'UNKNOWN',
      status:
        localStatus
    }
  );


  return {
    success: true,
    message: 'Status pembayaran diperbarui.',
    data: {
      bill_id:
        bill.bill_id,
      local_status:
        localStatus,
      engine_status:
        engineStatus,
      paid_at:
        patch.paid_at || bill.paid_at || ''
    }
  };
}


/* ============================================================
 * SECTION 7 — PAYMENT ENGINE CLIENT
 * ============================================================
 */

function createPaymentRequestToEngine_(
  input
) {

  const secret =
    getSppPaymentClientSecret_();


  const timestamp =
    new Date()
      .toISOString();


  const nonce =
    Utilities
      .getUuid()
      .replace(
        /-/g,
        ''
      );


  const payload = {

    source_app:
      SPP_DEMO_CONFIG.paymentSourceApp,

    reference_number:
      String(
        input.referenceNumber
      ).trim(),

    customer_id:
      String(
        input.customerId || ''
      ).trim(),

    customer_name:
      String(
        input.customerName
      ).trim(),

    description:
      String(
        input.description
      ).trim(),

    amount:
      Number(
        input.amount
      ),

    notes:
      String(
        input.notes || ''
      ).trim()
  };


  const payloadJson =
    JSON.stringify(
      payload
    );


  const action =
    'createPaymentRequest';


  const signature =
    createSppPaymentSignature_(
      action,
      SPP_DEMO_CONFIG.paymentClientId,
      timestamp,
      nonce,
      payloadJson,
      secret
    );


  const requestBody = {

    action:
      action,

    client_id:
      SPP_DEMO_CONFIG.paymentClientId,

    timestamp:
      timestamp,

    nonce:
      nonce,

    payload_json:
      payloadJson,

    signature:
      signature
  };


  const response =
    UrlFetchApp.fetch(
      SPP_DEMO_CONFIG.paymentEngineUrl,
      {
        method:
          'post',

        contentType:
          'application/json',

        payload:
          JSON.stringify(
            requestBody
          ),

        muteHttpExceptions:
          true
      }
    );


  return parsePaymentEngineResponse_(
    response
  );
}


function createSppPaymentSignature_(
  action,
  clientId,
  timestamp,
  nonce,
  payloadJson,
  secret
) {

  const rawToSign =
    'Action:' +
    action +
    '\n' +

    'Client-Id:' +
    clientId +
    '\n' +

    'Request-Timestamp:' +
    timestamp +
    '\n' +

    'Nonce:' +
    nonce +
    '\n' +

    'Payload:' +
    payloadJson;


  const signatureBytes =
    Utilities.computeHmacSha256Signature(
      rawToSign,
      secret,
      Utilities.Charset.UTF_8
    );


  return (
    'HMACSHA256=' +
    Utilities.base64Encode(
      signatureBytes
    )
  );
}


function parsePaymentEngineResponse_(
  response
) {

  const responseCode =
    response.getResponseCode();


  const responseText =
    response.getContentText();


  if (
    /^\s*<!DOCTYPE html>|^\s*<html/i.test(
      responseText
    )
  ) {

    throw new Error(
      'Payment Engine mengembalikan HTML Google, bukan JSON API.'
    );
  }


  let parsed;


  try {

    parsed =
      JSON.parse(
        responseText
      );

  } catch (error) {

    throw new Error(
      'Response Payment Engine bukan JSON valid.'
    );
  }


  if (
    responseCode !== 200 ||
    parsed.success !== true
  ) {

    throw new Error(
      parsed.message ||
      (
        'Payment Engine gagal. HTTP ' +
        responseCode
      )
    );
  }


  return parsed.data;
}


function getSecurePaymentStatusFromEngine_(
  token
) {

  const url =
    SPP_DEMO_CONFIG.paymentEngineUrl +
    '?action=getSecurePayment&token=' +
    encodeURIComponent(
      token
    ) +
    '&_=' +
    Date.now();


  const response =
    UrlFetchApp.fetch(
      url,
      {
        method:
          'get',
        muteHttpExceptions:
          true
      }
    );


  const responseText =
    response.getContentText();


  let parsed;


  try {

    parsed =
      JSON.parse(
        responseText
      );

  } catch (error) {

    throw new Error(
      'Status Payment Engine tidak dapat dibaca.'
    );
  }


  if (
    parsed.success !== true
  ) {

    throw new Error(
      parsed.message ||
      'Gagal membaca status pembayaran.'
    );
  }


  return parsed.data;
}

function createOrResetDemoSheets_(
  spreadsheet
) {

  const sheetDefinitions = {

    STUDENTS: [
      'student_id',
      'nis',
      'name',
      'class_name',
      'unit',
      'guardian_name',
      'guardian_phone',
      'status'
    ],

    BILLS: [
      'bill_id',
      'student_id',
      'period',
      'description',
      'amount',
      'due_date',
      'local_status',
      'payment_request_id',
      'payment_reference',
      'payment_link',
      'payment_token',
      'token_expires_at',
      'attempt_count',
      'paid_at',
      'last_checked_at',
      'created_at',
      'updated_at'
    ],

    ACTIVITY_LOG: [
      'logged_at',
      'action',
      'bill_id',
      'detail',
      'status'
    ]
  };


  Object.keys(
    sheetDefinitions
  ).forEach(
    function (sheetName) {

      let sheet =
        spreadsheet.getSheetByName(
          sheetName
        );


      if (!sheet) {

        sheet =
          spreadsheet.insertSheet(
            sheetName
          );
      }


      sheet.clear();


      const headers =
        sheetDefinitions[
          sheetName
        ];


      sheet
        .getRange(
          1,
          1,
          1,
          headers.length
        )
        .setValues(
          [
            headers
          ]
        );


      sheet
        .getRange(
          1,
          1,
          1,
          headers.length
        )
        .setFontWeight(
          'bold'
        );


      sheet.setFrozenRows(
        1
      );
    }
  );
}

function getStudentById_(
  spreadsheet,
  studentId
) {

  const sheet =
    spreadsheet.getSheetByName(
      SPP_SHEETS.students
    );


  const objects =
    sheetToObjects_(
      sheet
    );


  return (
    objects.find(
      function (row) {
        return (
          String(
            row.student_id
          ) ===
          String(
            studentId
          )
        );
      }
    ) ||
    null
  );
}


function getStudentBills_(
  spreadsheet,
  studentId
) {

  const sheet =
    spreadsheet.getSheetByName(
      SPP_SHEETS.bills
    );


  const objects =
    sheetToObjects_(
      sheet
    );


  return objects
    .filter(
      function (row) {
        return (
          String(
            row.student_id
          ) ===
          String(
            studentId
          )
        );
      }
    )
    .sort(
      function (a, b) {
        return (
          String(
            b.due_date
          ).localeCompare(
            String(
              a.due_date
            )
          )
        );
      }
    );
}

  const sheet =
    spreadsheet.getSheetByName(
      SPP_SHEETS.bills
    );


  const values =
    sheet.getDataRange()
      .getValues();


  if (
    values.length < 2
  ) {

    return null;
  }


  const headers =
    values[0];


  for (
    let rowIndex = 1;
    rowIndex < values.length;
    rowIndex++
  ) {

    const row =
      values[rowIndex];


    if (
      String(
        row[0]
      ) ===
      String(
        billId
      )
    ) {

      const data = {};


      headers.forEach(
        function (header, columnIndex) {
          data[
            header
          ] =
            normalizeSheetValue_(
              row[
                columnIndex
              ]
            );
        }
      );


      return {
        sheet:
          sheet,
        rowNumber:
          rowIndex + 1,
        headers:
          headers,
        data:
          data
      };
    }
  }


  return null;
}


function updateBillRow_(
  billResult,
  patch
) {

  const rowValues =
    billResult.headers.map(
      function (header) {

        if (
          Object.prototype
            .hasOwnProperty
            .call(
              patch,
              header
            )
        ) {

          return patch[
            header
          ];
        }


        return billResult.data[
          header
        ];
      }
    );


  billResult.sheet
    .getRange(
      billResult.rowNumber,
      1,
      1,
      rowValues.length
    )
    .setValues(
      [
        rowValues
      ]
    );
}


function sheetToObjects_(
  sheet
) {

  const values =
    sheet.getDataRange()
      .getValues();


  if (
    values.length < 2
  ) {

    return [];
  }


  const headers =
    values[0];


  return values
    .slice(
      1
    )
    .filter(
      function (row) {
        return row.some(
          function (cell) {
            return (
              cell !== '' &&
              cell !== null
            );
          }
        );
      }
    )
    .map(
      function (row) {

        const object = {};


        headers.forEach(
          function (header, index) {

            object[
              header
            ] =
              normalizeSheetValue_(
                row[
                  index
                ]
              );
          }
        );


        return object;
      }
    );
}


function normalizeSheetValue_(
  value
) {

  if (
    Object.prototype
      .toString
      .call(
        value
      ) ===
    '[object Date]'
  ) {

    return value.toISOString();
  }


  return value;
}


function calculateBillSummary_(
  bills
) {

  let totalOutstanding =
    0;


  let unpaidCount =
    0;


  let paidCount =
    0;


  bills.forEach(
    function (bill) {

      const status =
        String(
          bill.local_status || ''
        ).toUpperCase();


      if (
        status === 'PAID'
      ) {

        paidCount++;

      } else {

        totalOutstanding +=
          Number(
            bill.amount || 0
          );


        unpaidCount++;
      }
    }
  );


  return {
    total_outstanding:
      totalOutstanding,
    unpaid_count:
      unpaidCount,
    paid_count:
      paidCount,
    total_bills:
      bills.length
  };
}

function getSppPaymentClientSecret_() {

  const secret =
    String(
      PropertiesService
        .getScriptProperties()
        .getProperty(
          SPP_DEMO_CONFIG.paymentSecretProperty
        ) || ''
    ).trim();


  if (!secret) {

    throw new Error(
      SPP_DEMO_CONFIG.paymentSecretProperty +
      ' belum tersedia di Script Properties.'
    );
  }


  return secret;
}


function getOrCreateDemoRunId_(
  forceNew
) {

  const props =
    PropertiesService
      .getScriptProperties();


  let runId =
    String(
      props.getProperty(
        'SPP_DEMO_RUN_ID'
      ) || ''
    ).trim();


  if (
    forceNew ||
    !runId
  ) {

    runId =
      Utilities.formatDate(
        new Date(),
        SPP_DEMO_CONFIG.timezone,
        'yyyyMMddHHmmss'
      ) +
      Utilities
        .getUuid()
        .replace(
          /-/g,
          ''
        )
        .substring(
          0,
          4
        )
        .toUpperCase();


    props.setProperty(
      'SPP_DEMO_RUN_ID',
      runId
    );
  }


  return runId;
}


function extractTokenFromPaymentLink_(
  paymentLink
) {

  const link =
    String(
      paymentLink || ''
    );


  const match =
    link.match(
      /[#?&]token=([A-Fa-f0-9]{64,256})/
    );


  return match
    ? match[1]
    : '';
}


function outputSppDemoResponse_(
  payload,
  callback
) {

  const json =
    JSON.stringify(
      payload
    );


  const cleanCallback =
    String(
      callback || ''
    ).trim();


  if (
    cleanCallback &&
    /^[A-Za-z_$][0-9A-Za-z_$\.]{0,80}$/.test(
      cleanCallback
    )
  ) {

    return ContentService
      .createTextOutput(
        cleanCallback +
        '(' +
        json +
        ');'
      )
      .setMimeType(
        ContentService.MimeType.JAVASCRIPT
      );
  }


  return ContentService
    .createTextOutput(
      json
    )
    .setMimeType(
      ContentService.MimeType.JSON
    );
}
